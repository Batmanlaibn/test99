    <script>
// --- VARIABLES ---
const imageInput = document.getElementById("imageInput");
const userImage = document.getElementById("userImage");
const placeholder = document.getElementById("placeholder");
const placeholderText = document.getElementById("placeholderText");
const downloadBtn = document.getElementById("downloadBtn");

// Inputs de texto
const name1Input = document.getElementById("name1");
const name2Input = document.getElementById("name2");

// Elementos SVG
const bgColorCard = document.getElementById("bgCardSvg");
const svgName1 = document.getElementById("svgName1");
const svgName2 = document.getElementById("svgName2");
const phraseText = document.getElementById("phraseText");
const mainBorder = document.getElementById("mainBorder");
const svgElement = document.getElementById("cardSvg");
const canvas = document.getElementById("exportCanvas");
const headdingText = document.getElementById("headdingText");
const ampersand = document.getElementById("ampersand");

// --- FUNCIONES ---

// 1. Manejo de Imagen
imageInput.addEventListener("change", function (e) {
  const file = e.target.files[0];
  if (!file) return;

  // Validación del tipo de archivo
  if (!file.type.startsWith('image/')) {
    alert('Зөвхөн зургийн файл оруулна уу!');
    return;
  }

  const reader = new FileReader();
  reader.onload = function (event) {
    userImage.setAttribute("href", event.target.result);
    userImage.style.display = "block";
    placeholder.style.display = "none";
    placeholderText.style.display = "none";
    downloadBtn.disabled = false;
  };
  reader.onerror = function() {
    alert('Зураг уншихад алдаа гарлаа. Дахин оролдоно уу.');
  };
  reader.readAsDataURL(file);
});

// 2. Actualización en tiempo real de Nombres
function updateNames() {
  const name1 = name1Input.value.trim();
  const name2 = name2Input.value.trim();
  
  svgName1.textContent = name1 || "Нэр 1";
  svgName2.textContent = name2 || "Нэр 2";
}

name1Input.addEventListener("input", updateNames);
name2Input.addEventListener("input", updateNames);

// 3. Descarga (Mismo motor de renderizado)
downloadBtn.addEventListener("click", function () {
  // Deshabilitar el botón temporalmente para evitar clics múltiples
  downloadBtn.disabled = true;
  downloadBtn.textContent = "Боловсруулж байна...";
  
  setTimeout(() => {
    try {
      const width = 800;
      const height = 1000;
      canvas.width = width;
      canvas.height = height;
      const ctx = canvas.getContext("2d");

      const data = new XMLSerializer().serializeToString(svgElement);
      const svgBlob = new Blob([data], { type: "image/svg+xml;charset=utf-8" });
      const url = URL.createObjectURL(svgBlob);
      const img = new Image();

      img.onload = function () {
        ctx.drawImage(img, 0, 0, width, height);
        const pngUrl = canvas.toDataURL("image/png");
        const downloadLink = document.createElement("a");
        
        // Генерировать имя файла
        const name1 = name1Input.value.trim() || "хайр";
        const name2 = name2Input.value.trim() || "карт";
        downloadLink.href = pngUrl;
        downloadLink.download = `${name1}-${name2}-карт.png`;
        
        document.body.appendChild(downloadLink);
        downloadLink.click();
        document.body.removeChild(downloadLink);
        URL.revokeObjectURL(url);
        
        // Восстановить кнопку
        downloadBtn.disabled = false;
        downloadBtn.innerHTML = `<svg width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2">
          <path d="M21 15v4a2 2 0 0 1-2 2H5a2 2 0 0 1-2-2v-4"></path>
          <polyline points="7 10 12 15 17 10"></polyline>
          <line x1="12" y1="15" x2="12" y2="3"></line>
        </svg> PNG татах`;
      };
      
      img.onerror = function() {
        alert('Зураг татахад алдаа гарлаа. Дахин оролдоно уу.');
        downloadBtn.disabled = false;
        downloadBtn.innerHTML = `<svg width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2">
          <path d="M21 15v4a2 2 0 0 1-2 2H5a2 2 0 0 1-2-2v-4"></path>
          <polyline points="7 10 12 15 17 10"></polyline>
          <line x1="12" y1="15" x2="12" y2="3"></line>
        </svg> PNG татах`;
      };
      
      img.src = url;
    } catch (error) {
      console.error('Алдаа:', error);
      alert('Алдаа гарлаа. Дахин оролдоно уу.');
      downloadBtn.disabled = false;
      downloadBtn.innerHTML = `<svg width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2">
        <path d="M21 15v4a2 2 0 0 1-2 2H5a2 2 0 0 1-2-2v-4"></path>
        <polyline points="7 10 12 15 17 10"></polyline>
        <line x1="12" y1="15" x2="12" y2="3"></line>
      </svg> PNG татах`;
    }
  }, 100);
});
